Kandidat Model

# Tanpa Fourier
m1 <- Arima(train, order = c(1,0,2))
m2 <- Arima(train, order = c(2,0,1))
m3 <- Arima(train, order = c(2,0,2))
m4 <- Arima(train, order = c(5,0,0))

# Dengan Fourier
xr <- fourier(train, K = 3)
f1 <- Arima(train, order = c(1,0,2), xreg = xr)
f2 <- Arima(train, order = c(2,0,1), xreg = xr)
f3 <- Arima(train, order = c(2,0,2), xreg = xr)
f4 <- auto.arima(train, xreg = xr, seasonal = FALSE)
